class Writer:
    def __init__(self): self.data = bytearray()
    def write(self, value): self.data.extend(value)
    async def drain(self): pass
    def close(self): pass

class Socket:
    remote_address = ('127.0.0.1', 9999)
    open = True
    def __init__(self, messages=()): self.messages=iter(messages); self.sent=[]
    def __aiter__(self): return self
    async def __anext__(self):
        try: return next(self.messages)
        except StopIteration: raise StopAsyncIteration
    async def send(self, value): self.sent.append(value)

async def request(server, path, body=None):
    reader=asyncio.StreamReader(); writer=Writer()
    data=json.dumps(body).encode() if body is not None else b''
    method='POST' if body is not None else 'GET'
    reader.feed_data(f'{method} {path} HTTP/1.1\r\nHost: localhost\r\nContent-Length: {len(data)}\r\n\r\n'.encode()+data)
    reader.feed_eof()
    await server._handle_client(reader, writer)
    return bytes(writer.data).decode()
